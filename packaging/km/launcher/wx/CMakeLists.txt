cmake_minimum_required(VERSION 3.21)
project(KMSettings LANGUAGES C CXX RC)

if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Configure KMSettings with the x86 Microsoft C++ tools")
endif()
if(NOT EXISTS "${WX_WIDGETS_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set WX_WIDGETS_SOURCE_DIR to the extracted wxWidgets 3.2.12 source")
endif()

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
set(wxBUILD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
set(wxBUILD_INSTALL OFF CACHE BOOL "" FORCE)
set(wxBUILD_SAMPLES OFF CACHE STRING "" FORCE)
set(wxBUILD_TESTS OFF CACHE STRING "" FORCE)
set(wxUSE_GLCANVAS OFF CACHE BOOL "" FORCE)
set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
set(wxUSE_RICHTEXT OFF CACHE BOOL "" FORCE)
set(wxUSE_STC OFF CACHE BOOL "" FORCE)
set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
set(wxUSE_XRC OFF CACHE BOOL "" FORCE)
add_subdirectory("${WX_WIDGETS_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/wxwidgets")

add_library(km_settings_files STATIC ../settings_files.cpp)
target_include_directories(km_settings_files PUBLIC ..)
target_link_libraries(km_settings_files PUBLIC shell32 ole32)
target_compile_features(km_settings_files PUBLIC cxx_std_20)
target_compile_options(km_settings_files PRIVATE /utf-8 /W4)
target_compile_definitions(km_settings_files PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 WINVER=0x0601)

add_executable(KMSettings WIN32 wx_launcher.cpp launcher.rc)
target_include_directories(KMSettings BEFORE PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(KMSettings PRIVATE km_settings_files wx::core wx::base)
target_compile_options(KMSettings PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8>)
target_compile_definitions(KMSettings PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 WINVER=0x0601)
target_link_options(KMSettings PRIVATE /SUBSYSTEM:WINDOWS,6.01 /MANIFEST:NO)
set_target_properties(KMSettings PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

enable_testing()
add_executable(KMSettingsModelTest ../settings_test.cpp)
target_compile_features(KMSettingsModelTest PRIVATE cxx_std_20)
target_compile_options(KMSettingsModelTest PRIVATE /utf-8 /UNDEBUG)
add_test(NAME km-model COMMAND KMSettingsModelTest)

add_executable(KMSettingsFilesTest ../settings_files_test.cpp)
target_compile_options(KMSettingsFilesTest PRIVATE /utf-8 /UNDEBUG)
target_link_libraries(KMSettingsFilesTest PRIVATE km_settings_files)
add_test(NAME km-files COMMAND KMSettingsFilesTest)
